\begin{algorithm}
\caption{VerifyCasting(Instance, Certificate)}
\begin{algorithmic}[1]
\Require Instance containing $n$ (roles), $s$ (scenes), $k$ (actors), $AllowedActors[1..n]$, and $SceneRoles[1..s]$.
\Require Certificate array $Cast[1..n]$.
\Ensure \textbf{ACCEPT} if the certificate is valid, \textbf{REJECT} otherwise.
\Statex
\Comment{1. Verify that the certificate covers exactly $n$ roles}
\If{$\text{length}(Cast) \neq n$}
    \State \Return \textbf{REJECT}
\EndIf
\Statex
\State $p1\_participates \gets \text{FALSE}$
\State $p2\_participates \gets \text{FALSE}$
\Statex
\Comment{2. Verify Type 1 constraints and track actor participation}
\For{$i = 1 \dots n$}
    \State $actor \gets Cast[i]$
    \If{$actor \notin AllowedActors[i]$}
        \State \Return \textbf{REJECT}
    \EndIf
    \If{$actor == 1$} 
        \State $p1\_participates \gets \text{TRUE}$ 
    \EndIf
    \If{$actor == 2$} 
        \State $p2\_participates \gets \text{TRUE}$ 
    \EndIf
\EndFor
\Statex
\Comment{3. Verify both $p_1$ and $p_2$ participate in the play}
\If{\textbf{not} ($p1\_participates$ \textbf{and} $p2\_participates$)}
    \State \Return \textbf{REJECT}
\EndIf
\Statex
\Comment{4. Verify Type 2 constraints and scene exclusions}
\For{$j = 1 \dots s$}
    \State $p1\_in\_scene \gets \text{FALSE}$
    \State $p2\_in\_scene \gets \text{FALSE}$
    \State $ActorsInCurrentScene \gets \emptyset$
    \Statex
    \For{$role \in SceneRoles[j]$}
        \State $actor \gets Cast[role]$
        \If{$actor \in ActorsInCurrentScene$}
            \State \Return \textbf{REJECT} \Comment{One actor cannot play multiple roles in a scene}
        \EndIf
        \State $ActorsInCurrentScene.\text{add}(actor)$
        \Statex
        \If{$actor == 1$} 
            \State $p1\_in\_scene \gets \text{TRUE}$ 
        \EndIf
        \If{$actor == 2$} 
            \State $p2\_in\_scene \gets \text{TRUE}$ 
        \EndIf
    \EndFor
    \Statex
    \If{$p1\_in\_scene$ \textbf{and} $p2\_in\_scene$}
        \State \Return \textbf{REJECT} \Comment{$p_1$ and $p_2$ cannot be in the same scene}
    \EndIf
\EndFor
\Statex
\State \Return \textbf{ACCEPT}
\end{algorithmic}
\end{algorithm}
